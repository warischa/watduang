doc.fire('pointerdown',{timeStamp:0});
const cancelOuter=armAllButtons(outer); if(INIT==='disabled') shared.disabled=false;
tick(450); const q1=shared.disabled;
outer.dispatch('pointerdown',{timeStamp:200}); const reHeld=shared.disabled;
tick(150); const cancelInner=armAllButtons(inner);

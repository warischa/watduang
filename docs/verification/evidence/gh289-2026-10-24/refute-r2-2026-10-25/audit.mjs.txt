class Element {
  constructor(id = '') {
    this.id = id; this.style = {}; this.attributes = []; this.children = [];
    this.events = new Map(); this.classes = new Set(); this.textContent = '';
    this.clientWidth = 350; this.checked = false; this.value = ''; this.disabled = false;
    this.classList = { add: c => this.classes.add(c), remove: c => this.classes.delete(c) };
  }
  addEventListener(type, fn) { if (!this.events.has(type)) this.events.set(type, new Set()); this.events.get(type).add(fn); }
  removeEventListener(type, fn) { this.events.get(type)?.delete(fn); }
  fire(type, fields = {}) { for (const fn of [...(this.events.get(type) || [])]) fn({ target: this, ...fields }); }
  appendChild(el) { this.children.push(el); return el; }
  removeChild(el) { this.children.splice(this.children.indexOf(el), 1); }
  get firstChild() { return this.children[0]; }
  setAttribute(name, value) { this[name] = value; }
  getBoundingClientRect() { return { width: 350 }; }
}
function wheel(ref, reduced = false) {
  const elements = Object.fromEntries(['wheel-disc', 'wheel-group', 'wheel-result', 'wheel-note', 'wheel-names', 'wheel-spin', 'wheel-stop', 'wheel-reset', 'wheel-eliminate'].map(id => [id, new Element(id)]));
  const auto = new Element(); auto.checked = true; auto.value = 'auto';
  const press = new Element(); press.value = 'press';
  const document = new Element();
  document.getElementById = id => elements[id];
  document.querySelector = selector => elements[selector.slice(1)];
  document.querySelectorAll = () => [auto, press];
  document.createElement = () => new Element();
  document.createElementNS = () => new Element();
  let now = 0, next = 0;
  const timers = new Map();
  const window = { matchMedia: () => ({ matches: reduced }), setTimeout: (fn, delay) => { const id = ++next; timers.set(id, { fn, at: now + delay }); return id; } };
  const clearTimeout = id => timers.delete(id);
  const script = sourceAt(ref, 'src/pages/tool/wheel.astro').split('<script>')[1].split('</script>')[0];
  const bundled = esbuild.buildSync({ stdin: { contents: script, resolveDir: repo + '/src/pages/tool', loader: 'ts' }, bundle: true, write: false, format: 'iife', platform: 'browser' }).outputFiles[0].text;
  vm.runInNewContext(bundled, { document, window, clearTimeout, crypto: webcrypto, getComputedStyle: () => ({ transform: 'none' }), console });
  document.fire('watduang:start', { detail: { players: ['A', 'B', 'C', 'D', 'E'] } });
  const advance = until => {
    while (true) {
      const due = [...timers].filter(([, t]) => t.at <= until).sort((a, b) => a[1].at - b[1].at)[0];
      if (!due) break;
      now = due[1].at; timers.delete(due[0]); due[1].fn();
    }
    now = until;
  };
  return { elements, auto, press, document, advance, now: () => now, timers };
}

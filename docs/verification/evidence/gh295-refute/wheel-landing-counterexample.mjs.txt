async function bundle(mode, path) {
  const code = source(mode, path).match(/<script>([\s\S]*?)<\/script>/)[1];
  const out = await build({
    stdin: { contents: code, loader: 'ts', resolveDir: repo + '/' + path.slice(0, path.lastIndexOf('/')), sourcefile: path },
    bundle: true, write: false, format: 'iife', platform: 'browser',
    plugins: mode === 'base' ? [{ name: 'head-sources', setup(b) {
      b.onLoad({ filter: /\.ts$/ }, (args) => ({ contents: source(mode, args.path.slice(repo.length + 1)), loader: 'ts' }));
    } }] : [],
  });
  return out.outputFiles[0].text;
}

function clock() {
  let now = 0, id = 0;
  const timers = new Map();
  return {
    now: () => now,
    setTimeout(fn, ms) { const key = ++id; timers.set(key, { fn, at: now + ms }); return key; },
    clearTimeout(key) { timers.delete(key); },
    tick(ms) {
      const end = now + ms;
      for (;;) {
        const next = [...timers].filter(([, t]) => t.at <= end).sort((a, b) => a[1].at - b[1].at)[0];
        if (!next) break;
        now = next[1].at; timers.delete(next[0]); next[1].fn();
      }
      now = end;
    },
  };
}

FakeElement.prototype.getBoundingClientRect = function () { return { width: 360, height: 360 }; };
FakeElement.prototype.scrollIntoView = function (opts) { this.lastScroll = opts; };
Object.defineProperty(FakeElement.prototype, 'attributes', { get() { return Object.entries(this._attrs).map(([name, value]) => ({ name, value })); } });
Object.defineProperty(FakeElement.prototype, 'firstChild', { get() { return this.children[0] ?? null; } });
FakeElement.prototype.removeChild = function (kid) { this.children.splice(this.children.indexOf(kid), 1); };
Object.defineProperty(FakeElement.prototype, 'classList', { get() {
  const node = this;
  return { add(c) { node.className = [...new Set([...node.className.split(/\s+/).filter(Boolean), c])].join(' '); },
    remove(c) { node.className = node.className.split(/\s+/).filter(x => x !== c).join(' '); } };
} });

async function setup(mode, reduced = false) {
  const doc = makeDocument();
  const time = clock();
  const els = {};
  const add = (id, tag = 'button', parent) => {
    const el = new FakeElement(tag, doc); el.id = id; els[id] = el;
    parent?.appendChild(el); return el;
  };
  const section = add('wheel', 'section');
  add('wheel-disc', 'button', section);
  add('wheel-group', 'g', els['wheel-disc']);
  for (const id of ['wheel-result', 'wheel-note', 'wheel-names']) add(id, 'div', section);
  for (const id of ['wheel-spin', 'wheel-stop', 'wheel-reset']) add(id, 'button', section);
  add('wheel-eliminate', 'input', section).checked = false;
  const root = add('tool-name-entry', 'div', section); root.dataset = { storageKey: 'judge:names' };
  add('name-input', 'textarea', root).value = '';
  add('name-start', 'button', root);
  const auto = add('mode-auto', 'input', section); auto.value = 'auto'; auto.checked = true;
  const press = add('mode-press', 'input', section); press.value = 'press'; press.checked = false;
  doc.getElementById = id => els[id];
  doc.querySelector = sel => els[sel.slice(1)];
  doc.querySelectorAll = () => [auto, press];
  doc.createElementNS = (_, tag) => new FakeElement(tag, doc);
  doc.dispatchEvent = ev => { doc.fire(ev.type, ev); return true; };
  const store = new Map();
  const context = vm.createContext({
    document: doc, window: { matchMedia: () => ({ matches: reduced }), setTimeout: time.setTimeout },
    localStorage: { getItem: key => store.get(key) ?? null, setItem: (key, val) => store.set(key, val) },
    CustomEvent: class { constructor(type, props) { this.type = type; Object.assign(this, props); } },
    setTimeout: time.setTimeout, clearTimeout: time.clearTimeout, console, Intl, Math,
    getComputedStyle: () => ({ transform: "none" }),
  });
  vm.runInContext(await bundle(mode, 'src/components/ToolNameEntry.astro'), context);
  vm.runInContext(await bundle(mode, 'src/pages/tool/wheel.astro'), context);
  const contact = () => {
    const ev = { timeStamp: time.now() };
    doc.fire('pointerdown', ev); section.dispatch('pointerdown', ev);
    doc.fire('pointerup', { timeStamp: time.now() + 1 });
  };
  return { els, time, doc, start(names) { els['name-input'].value = names; contact(); els['name-start'].click(); },
    modePress() { contact(); auto.checked = false; press.checked = true; press.dispatch('change'); },
    modeAuto() { contact(); auto.checked = true; press.checked = false; auto.dispatch('change'); },
    tap(id) { contact(); els[id].click(); },
  };
}



const rows = [];
for (const mode of ['base', 'current']) {
  const s = await setup(mode);
  s.start('A\nB\nC');
  s.time.tick(450);
  s.els['wheel-spin'].click();
  s.time.tick(5800); // now 6250; spin's landing fallback is due at 6450
  s.els.wheel.dispatch('pointerdown', { timeStamp: 200 }); // re-hold until 6650
  const afterRehold = s.els['wheel-spin'].disabled;
  s.time.tick(250); // now 6500: the spin has landed, gate still has 150ms left
  const afterLanding = s.els['wheel-spin'].disabled;
  const before = s.els['wheel-group'].style.transform;
  const resultsBeforeTap = s.els['wheel-result'].children.length;
  s.tap('wheel-spin'); // still inside active quiet window
  rows.push({mode, afterRehold, afterLanding, resultsBeforeTap,
    secondSpin: s.els['wheel-group'].style.transform !== before,
    note: s.els['wheel-note'].textContent});
}
console.log(JSON.stringify(rows, null, 2));
assert.equal(rows[1].afterRehold, true, 'positive control: current does hold spin when re-entering the gate');
assert.equal(rows[1].afterLanding, true, 'current: landing must not release spin inside the re-entered quiet window');

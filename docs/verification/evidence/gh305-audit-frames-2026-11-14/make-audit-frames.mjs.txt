// Saved as .txt so repo tooling never treats it as code. Run from a copy OUTSIDE the repo:
//   cp make-audit-frames.mjs.txt <tmp>/gh305-af/make-audit-frames.mjs
//   node <tmp>/gh305-af/make-audit-frames.mjs <repo root> write [measurements.json]   # captions take numbers from a prior run

// fire; the as-built HUD line does not cross rail 1 or the wrapped one does; a face is not loaded; a dotted circle
// (U+25CC) is in the rendered text; innerWidth is not the width set; a caption number differs from the fresh measurement.

// Today's layout (fieldLayout in the cards renderer's plan): the board card at 100-card metrics, the whole field
// scaled by one factor with the column count 6 to 10 that gives the largest factor.
const todayLayout = (n) => {
  let best = { cols: 6, s: 0 };
  for (let cols = 6; cols <= 10; cols++) {
    const rows = Math.max(1, Math.ceil(n / cols));
    const w = cols * CARD.w + (cols - 1) * CARD.colGap;
    const h = rows * CARD.h + (rows - 1) * CARD.rowGap;
    const s = Math.min(1, (FIELD.right - FIELD.left) / w, FIELD.height / h);
    if (s > best.s + 1e-9) best = { cols, rows, s, w, h };
  }
  return best;
};

const api = async (p, method = 'GET') => (await fetch(`http://127.0.0.1:${PORT}${p}`, { method })).json();
const open = async () => {
  const target = await api('/json/new?about:blank', 'PUT');
  const ws = new WebSocket(target.webSocketDebuggerUrl);
  let id = 0; const pending = new Map(); const errors = []; let loaded = null;
  const send = (method, params = {}) => new Promise((res) => { pending.set(++id, res); ws.send(JSON.stringify({ id, method, params })); });
  ws.addEventListener('message', (ev) => {
    const m = JSON.parse(ev.data);
    if (m.id && pending.has(m.id)) { pending.get(m.id)(m); pending.delete(m.id); return; }
    if (m.method === 'Page.loadEventFired' && loaded) { loaded(); loaded = null; }
    if (m.method === 'Runtime.exceptionThrown') errors.push(m.params.exceptionDetails?.exception?.description ?? m.params.exceptionDetails?.text);
    if (m.method === 'Log.entryAdded' && m.params.entry.level === 'error') errors.push(`${m.params.entry.text} ${m.params.entry.url ?? ''}`.trim());
  });
  await new Promise((r) => ws.addEventListener('open', r));
  await send('Page.enable'); await send('Runtime.enable'); await send('Log.enable');
  const nav = async (url, w, h) => {
    // The protocol call that genuinely sets the viewport (browser-verification.md, trap 1); innerWidth is checked.
    await send('Emulation.setDeviceMetricsOverride', { width: w, height: h, deviceScaleFactor: 1, mobile: false });
    const lp = new Promise((r) => { loaded = r; }); await send('Page.navigate', { url }); await lp;
  };
  const evaluate = async (expression) => (await send('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true })).result?.result?.value;
  const close = async () => { await fetch(`http://127.0.0.1:${PORT}/json/close/${target.id}`); ws.close(); };
  return { send, nav, evaluate, close, errors };
};

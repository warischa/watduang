async function shotsDuring(T, tag, label, sel, { from = 700, step = 450, n = 5, maxMs = 16000 } = {}) {
  await T.tapEl(sel, label, { settleMs: 450 });
  const t0 = Date.now(); let saw = false; const shots = []; let k = 0;
  while (Date.now() - t0 < maxMs) {
    const c = await T.ev(`return !!document.querySelector('.pr-3d canvas, .pd-3d canvas')`);
    if (c) saw = true;
    if (saw && !c) break;
    if (c && Date.now() - t0 >= from + k * step && k < n) { shots.push({ ms: Date.now() - t0, file: await T.shot(`${tag}-${String(k).padStart(2, '0')}`) }); k++; }
    await sleep(40);
  }
  return { saw, shots };
}
async function dealShots(T, tag, label, sel) {
  const r = {}; r.ref = await T.ev(`return ${REF}`);
  r.deal = await shotsDuring(T, `${tag}-deal`, label, sel, { from: 3000, step: 150, n: 8 });
  await sleep(600);
  await T.ev(`window.__wd.texts.length=0; return 1`);
  r.reveal = await shotsDuring(T, `${tag}-reveal`, 'เปิดผล', sel, { from: 150, step: 300, n: 8 });
  r.fonts = await T.ev(`const t=window.__wd.texts.filter(x=>/^\d{3}$/.test(x.s)); const f={}; for(const x of t){ const k=x.font+' @'+x.cw+'x'+x.ch; f[k]=(f[k]||0)+1; } const names=[...new Set(window.__wd.texts.map(x=>x.font+' @'+x.cw+'x'+x.ch))].slice(0,12); return {digitTexts:t.length,byFont:f,allFonts:names}`);
  return r;
}

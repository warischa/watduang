import sys,json
import numpy as np
from PIL import Image
sys.path.insert(0,'<scratch>/work')
import cardgrid
EV='<repo>/docs/verification/evidence/gh305-audit-measure-2026-11-14/shots/'
def white_bbox(im,x0,y0,x1,y1,thr=170):
    sub=im[y0:y1,x0:x1]; m=(sub[...,0]>thr)&(sub[...,1]>thr)&(sub[...,2]>thr)
    ys=np.where(m.any(axis=1))[0]; xs=np.where(m.any(axis=0))[0]
    if len(ys)==0: return None
    return (xs[0]+x0,ys[0]+y0,xs[-1]-xs[0]+1,ys[-1]-ys[0]+1)
def cluster(vals,tol):
    vals=sorted(vals); out=[[vals[0]]]
    for v in vals[1:]:
        if v-out[-1][-1]<=tol: out[-1].append(v)
        else: out.append([v])
    return out
def run(png,tag=None,scale=0.9481,ox=49.78,oy=56):
    im=np.array(Image.open(EV+png).convert('RGB')).astype(int)
    boxes=cardgrid.analyse(EV+png,(0,140,1920,880),scale,ox,oy)
    cols=cluster([b[0] for b in boxes],25); rows=cluster([b[1] for b in boxes],12)
    res={'png':png,'cards':len(boxes),'cols':len(cols),'rows':len(rows)}
    # reference overlay: lines at y 766..790 / 790..812 (400 then 700) x 206..300
    ref400=white_bbox(im,200,760,306,787); ref700=white_bbox(im,200,788,306,816)
    res['ref400_bbox']=ref400; res['ref700_bbox']=ref700
    # glyph of the first card of the first row
    first=sorted(boxes,key=lambda b:(round(b[1]/12),b[0]))[0]
    x,y,w,h=first
    g=white_bbox(im,x+6,y+4,x+w-6,y+h-4)
    res['card1_box']=first; res['card1_glyph_bbox']=g
    # glyph heights across the first row of cards
    row=[b for b in boxes if abs(b[1]-first[1])<10]; gh=[]
    for (x,y,w,h) in row:
        gg=white_bbox(im,x+6,y+4,x+w-6,y+h-4)
        if gg: gh.append(gg[3])
    res['row0_glyph_h']=sorted(set(gh))
    if ref700 and g: res['painted_name_px_vs_700ref16']=round(16*g[3]/ref700[3],2)
    if ref400 and g: res['painted_name_px_vs_400ref16']=round(16*g[3]/ref400[3],2)
    res['card_frame_px']=[round(w/scale,1),round(h/scale,1)]
    return res
if __name__=='__main__':
    for a in sys.argv[1:]: print(json.dumps(run(a),default=lambda o:int(o) if hasattr(o,'item') else str(o)))

async function seed(T, { names, prizes = 3, sched = null, games = {} }) {
  await openSetup(T, names, { event: 'Audit' });
  await setPrizes(T, prizes);
  if (sched) for (let i = 0; i < 3; i++) await setIn(T, `pd-s${i + 1}`, sched[i]);
  for (const [g, v] of [['pd-game-r1', games.r1 ?? null], ['pd-game-r2', games.r2 ?? null], ['pd-game-r3', null], ['pd-game-r4', null], ['pd-game-final', games.final ?? null]]) await pickGame(T, g, v);
  const arrived = await startEvent(T);
  return arrived;
}
async function toWidth(T, W) { const mobile = MOB ?? W < 640; await T.view(W, H(W), mobile); await T.nav(`${BASE}/game/prize-draw/play/`); await T.view(W, H(W), mobile); await sleep(900); return { inner: await T.ev('return [innerWidth, document.documentElement.scrollWidth]') }; }
async function rounds2D(T, n) { for (let i = 0; i < n; i++) { await T.tapEl('.pd-mc button', 'เปิดผล', { settleMs: 450 }); await sleep(1300); await T.tapEl('.pd-mc button', 'รอบต่อไป'); await sleep(900); } }
async function observe(T, label, tag, { maxMs = 15000, afterSawMs = 2500, shotAt = [1500] } = {}) {
  const req0 = T.S.requests.length; const c0 = await T.ev('return __wd.ctx.filter(c=>c.ok).length');
  await T.tapEl('.pd-mc button', label, { settleMs: 450 });
  const t0 = Date.now(); let saw = false, sawAt = null, shots = []; let endBtn = null;
  while (Date.now() - t0 < maxMs) {
    const s = await T.ev(`const c=document.querySelector('.pd-3d canvas'); const b=document.querySelector('.pd-mc button'); return {c:!!c,cw:c?c.getBoundingClientRect().width:0,btn:b?b.textContent.trim():null,ph:(window.__watduangRevealFrameTime.frames||[]).length}`);
    endBtn = s.btn;
    if (s.c && !saw) { saw = true; sawAt = Date.now() - t0; }
    if (saw) { const since = Date.now() - t0 - sawAt; for (const sa of shotAt) if (since >= sa && !shots.includes(sa)) { shots.push(sa); await T.shot(`${tag}-3d-${sa}`); } if (since >= afterSawMs) break; }
    if (!saw && s.btn && s.btn !== label && Date.now() - t0 > 2500) break;
    await sleep(100);
  }
  const reqs = T.S.requests.slice(req0).map((u) => u.split('/').pop());
  const c1 = await T.ev('return __wd.ctx.filter(c=>c.ok).length');
  if (!saw) await T.shot(`${tag}-2d`);
  return { label, sawCanvas: saw, sawAtMs: sawAt, endBtn, newGlContexts: c1 - c0, threeChunkSincePress: reqs.some((u) => /three\.module/.test(u)), threeChunkEver: T.S.requests.some((u) => /three\.module/.test(u)), jsSincePress: reqs.filter((u) => /\.js$/.test(u)).length };
}

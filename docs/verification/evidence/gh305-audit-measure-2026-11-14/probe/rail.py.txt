import json,sys
import numpy as np
from PIL import Image
def lum(c):
    c=np.array(c,dtype=float)/255
    c=np.where(c<=0.03928,c/12.92,((c+0.055)/1.055)**2.4)
    return 0.2126*c[...,0]+0.7152*c[...,1]+0.0722*c[...,2]
def ratio(a,b):
    la,lb=lum(a),lum(b); hi,lo=max(la,lb),min(la,lb); return (hi+0.05)/(lo+0.05)
def hudgap(png,lines,pad=3):
    im=np.array(Image.open(png).convert('RGB')).astype(int)
    out=[]
    for ln in lines:
        r=int(ln['r']); res=[]
        for y in np.linspace(ln['t']+4,ln['b']-4,9).astype(int):
            row=im[y]; g=row[r+6]; 
            d=np.abs(row-g).sum(axis=1)
            xs=[x for x in range(r+6,min(r+700,im.shape[1])) if d[x]>40]
            res.append(xs[0]-ln['r'] if xs else None)
        out.append({'right':round(ln['r'],1),'gaps_by_row':res,'min':min([v for v in res if v is not None],default=None)})
    return out
def railcontrast(png,ys,xwin):
    im=np.array(Image.open(png).convert('RGB')).astype(int)
    res=[]
    for y,(x0,x1) in zip(ys,xwin):
        brights=[];rx=[]
        for yy in range(y-2,y+3):
            row=im[yy,x0:x1]; L=lum(row); idx=np.argsort(L)[-3:]; brights+= [row[i] for i in idx]; rx+= [x0+i for i in idx]
        rail=np.mean(brights,axis=0); x=int(np.median(rx))
        ground=im[y-2:y+3,x+40-2:x+40+3].reshape(-1,3).mean(axis=0)
        res.append({'y':y,'railx':x,'rail':[round(v) for v in rail],'ground':[round(v) for v in ground],'ratio':round(float(ratio(rail,ground)),2)})
    return res

async function openTab({ noGL = false, reducedMotion = false } = {}) {
  const target = await (await fetch(`http://127.0.0.1:${CDP_PORT}/json/new?about:blank`, { method: 'PUT' })).json();
  const ws = new WebSocket(target.webSocketDebuggerUrl);
  let id = 0;
  const pending = new Map();
  const S = { console: [], exceptions: [], log: [], requests: [], dialogs: [], lostEvents: [] };
  let loadResolve = null;
  ws.addEventListener('message', (e) => {
    const m = JSON.parse(e.data);
    if (m.id && pending.has(m.id)) { pending.get(m.id)(m); pending.delete(m.id); return; }
    const p = m.params;
    if (m.method === 'Page.loadEventFired' && loadResolve) { loadResolve(); loadResolve = null; }
    if (m.method === 'Runtime.consoleAPICalled') S.console.push({ type: p.type, text: (p.args || []).map((a) => a.value ?? a.description ?? '').join(' ').slice(0, 400), url: p.stackTrace?.callFrames?.[0]?.url?.slice(-60) });
    if (m.method === 'Runtime.exceptionThrown') S.exceptions.push((p.exceptionDetails?.exception?.description ?? p.exceptionDetails?.text ?? '').split('\n')[0].slice(0, 400));
    if (m.method === 'Log.entryAdded') S.log.push({ level: p.entry.level, source: p.entry.source, text: p.entry.text.slice(0, 400), url: (p.entry.url || '').slice(-60) });
    if (m.method === 'Network.requestWillBeSent') S.requests.push(p.request.url);
    if (m.method === 'Page.javascriptDialogOpening') {
      S.dialogs.push({ type: p.type, message: p.message.slice(0, 80) });
      send('Page.handleJavaScriptDialog', { accept: true });
    }
  });
  await new Promise((r) => ws.addEventListener('open', r));
  const send = (method, params = {}) => new Promise((res) => { pending.set(++id, res); ws.send(JSON.stringify({ id, method, params })); });
  for (const d of ['Page', 'Runtime', 'Log', 'Network']) await send(`${d}.enable`);
  await send('Network.setCacheDisabled', { cacheDisabled: true });
  await send('Page.addScriptToEvaluateOnNewDocument', { source: initSource(noGL) });
  if (reducedMotion) await send('Emulation.setEmulatedMedia', { features: [{ name: 'prefers-reduced-motion', value: 'reduce' }] });
  const T = {
    S, send,
    async nav(url) { const p = new Promise((r) => { loadResolve = r; }); await send('Page.navigate', { url }); await p; await sleep(900); },
    async ev(body) {
      const r = await send('Runtime.evaluate', { expression: `(async () => { ${body} })()`, awaitPromise: true, returnByValue: true });
      if (r.result?.exceptionDetails) return { __error: r.result.exceptionDetails.exception?.description ?? r.result.exceptionDetails.text };
      return r.result?.result?.value;
    },
    async view(w, h, mobile = false) { await send('Emulation.setDeviceMetricsOverride', { width: w, height: h, deviceScaleFactor: 1, mobile }); await sleep(300); },
    async shot(name, { clip } = {}) {
      await send('Page.bringToFront');
      const r = await send('Page.captureScreenshot', { format: 'png', ...(clip ? { clip } : {}) });
      writeFileSync(join(OUT, 'shots', name + '.png'), Buffer.from(r.result.data, 'base64'));
      return name + '.png';
    },
    async touch(x, y) {
      await send('Page.bringToFront');
      await send('Input.dispatchTouchEvent', { type: 'touchStart', touchPoints: [{ x, y }] });
      await send('Input.dispatchTouchEvent', { type: 'touchEnd', touchPoints: [] });
    },
    async center(sel, text) {
      return T.ev(`const els=[...document.querySelectorAll(${JSON.stringify(sel)})].filter(e=>!${JSON.stringify(text ?? null)}||e.textContent.trim()===${JSON.stringify(text ?? null)});
        const e=els[0]; if(!e) return null; e.scrollIntoView({block:'center'}); const r=e.getBoundingClientRect(); const x=r.left+r.width/2,y=r.top+r.height/2;
        const hit=document.elementFromPoint(x,y); return {x,y,disabled:!!e.disabled,hit:!!hit&&(hit===e||e.contains(hit)),text:e.textContent.trim().slice(0,30)};`);
    },
    async waitFor(expr, ms = 15000, step = 100) {
      const t0 = Date.now();
      while (Date.now() - t0 < ms) { const v = await T.ev(`return (${expr});`); if (v && !v.__error) return v; await sleep(step); }
      return null;
    },
    /** tap an enabled control found by selector (+exact text); returns the pre-tap reading. */
    async tapEl(sel, text, { waitMs = 15000, settleMs = 150 } = {}) {
      const t0 = Date.now();
      let c = null;
      while (Date.now() - t0 < waitMs) {
        c = await T.center(sel, text);
        if (c && !c.disabled) break;
        await sleep(80);
      }
      if (!c || c.disabled) throw new Error(`tapEl: ${sel} ${text ?? ''} not enabled in ${waitMs}ms (${JSON.stringify(c)})`);
      await sleep(settleMs);
      c = await T.center(sel, text);
      if (!c.hit) throw new Error(`tapEl: ${sel} ${text ?? ''} covered at its centre`);
      await T.touch(c.x, c.y);
      return c;
    },
    async close() { await fetch(`http://127.0.0.1:${CDP_PORT}/json/close/${target.id}`); ws.close(); },
  };
  await T.view(1920, 1080);
  return T;
}

/** pick the finalist `which` (0 = first listed) onto lane `lane`; the lane tap, then the name tap. */
async function pickLane(T, lane) {
  await sleep(450);
  let c = await centerN(T, '.pd-lane', lane);
  let blocked = null;
  if (!c) throw new Error(`lane ${lane} absent`);
  if (!c.hit) {
    // record what covers the lane control's centre, then look for any point of the control that does hit it
    blocked = await T.ev(`const e=[...document.querySelectorAll('.pd-lane')][${lane}]; const r=e.getBoundingClientRect(); const top=document.elementFromPoint(${c.x},${c.y}); const d=x=>x?(x.tagName.toLowerCase()+(x.id?'#'+x.id:'')+(x.className&&x.className.baseVal===undefined?'.'+String(x.className).split(' ').join('.'):'')):null; let free=null; for(let fy=0.1;fy<=0.9&&!free;fy+=0.2) for(let fx=0.1;fx<=0.9&&!free;fx+=0.2){const x=r.left+r.width*fx,y=r.top+r.height*fy; const h=document.elementFromPoint(x,y); if(h&&(h===e||e.contains(h))) free={x,y};} return { rect:{l:Math.round(r.left),t:Math.round(r.top),w:Math.round(r.width),h:Math.round(r.height)}, coveredBy:d(top), coveredByRect: top?(()=>{const q=top.getBoundingClientRect(); return {l:Math.round(q.left),t:Math.round(q.top),w:Math.round(q.width),h:Math.round(q.height)}})():null, free };`);
    if (blocked.free) c = { ...c, x: blocked.free.x, y: blocked.free.y };
    else { await T.ev(`[...document.querySelectorAll('.pd-lane')][${lane}].click(); return 1`); }
  }
  if (!blocked || blocked.free) await T.touch(c.x, c.y);
  await sleep(300);
  const nm = await T.ev(`const b=document.querySelector('.pd-name'); return b?b.textContent.trim():null`);
  const n = await T.tapEl('.pd-name', null, { settleMs: 450 });
  await sleep(300);
  return { lane, name: nm, hit: n.hit, blocked };
}

/** one press with screenshots while a 3D canvas is up; names carry the host's own phase. */
async function pressShots(T, label, prefix, { sel = '.pd-mc button', canvasSel = '.pd-3d canvas', everyMs = 2000, maxShots = 6, maxMs = 150000 } = {}) {
  const w0 = await wd2(T); const t0 = Date.now();
  const tap = await T.tapEl(sel, label, { settleMs: 450 });
  const shots = [], tl = []; let saw = false, last = -1e9, done = false;
  while (Date.now() - t0 < maxMs) {
    const r = await T.ev(`const c=document.querySelector(${JSON.stringify(canvasSel)}); const b=document.querySelector(${JSON.stringify(sel)}); return { canvas:!!c, btn:b?b.textContent.trim():null };`);
    const ms = Date.now() - t0;
    const ph = r.canvas ? await phaseNow(T) : '-';
    if (!tl.length || tl[tl.length - 1].c !== r.canvas || tl[tl.length - 1].b !== r.btn || tl[tl.length - 1].p !== ph) tl.push({ ms, c: r.canvas, b: r.btn, p: ph });
    if (r.canvas) {
      saw = true;
      if (shots.length < maxShots && ms - last >= everyMs) { last = ms; shots.push({ ms, phase: ph, file: await T.shot(`${prefix}-${nn(shots.length + 1)}-${ph}`) }); }
    }
    if (!r.canvas && saw) { await sleep(700); done = true; break; }
    if (!r.canvas && r.btn && r.btn !== label && ms > 4000) { done = true; break; }
    await sleep(80);
  }
  const w1 = await wd2(T);
  return { tap: { disabledAtTap: tap.disabled, hit: tap.hit }, sawCanvas: saw, done, elapsedMs: Date.now() - t0, tl: tl.slice(0, 40), shots, canvasEvents: w1.canvas.length - w0.canvas.length, frames: w1.frames, finalBtn: await mcText(T) };
}

    R.pickR3 = await pickGame(T, 'pd-game-r3', 'cards'); R.pickR4 = await pickGame(T, 'pd-game-r4', 'cards');
    R.afterPicks = { pressed: await pressedRead(T), cards: await cardStates(T) };
    await T.ev(`document.getElementById('pd-game-r1').scrollIntoView({block:'center'}); return 1`);
    await T.shot('game-prize-draw-1920-cards-picker-halving');
    // a schedule edit that stops round 1 halving must clear its cards pick and disable the button
    await setIn(T, 'pd-s1', 25); await sleep(300);
    R.edited = { sched: await schedRead(T), cards: await cardStates(T), pressed: await pressedRead(T) };
    await T.shot('game-prize-draw-1920-cards-picker-r1-not-halving');
    await setIn(T, 'pd-s1', 20); await sleep(300);
    R.restored = { sched: await schedRead(T), cards: await cardStates(T), pressed: await pressedRead(T) };
    R.rePickR1 = await pickGame(T, 'pd-game-r1', 'cards'); R.rePickR2 = await pickGame(T, 'pd-game-r2', 'cards');
    R.finalPressed = await pressedRead(T);
    R.arrived = await startEvent(T);
    R.rounds = [];
    for (const round of [1, 2]) {
      const before = await readEvent(T);
      const label = await mcText(T);
      const deal = await pressShots(T, 'แจกไพ่', `b3-r${round}-deal`, { everyMs: 3000, maxShots: 2 });
      const reveal = await pressShots(T, 'เปิดผล', `b3-r${round}-reveal`, { everyMs: 3000, maxShots: 3 });
      const e = await readEvent(T); const c = await readChips(T);
      const rec = e.rounds[round - 1];
      R.rounds.push({ round, firstLabel: label, liveBefore: before.nRounds === round - 1 ? (round === 1 ? 40 : e.rounds[0].n) : null, deal: { saw: deal.sawCanvas, done: deal.done, shots: deal.shots.length }, reveal: { saw: reveal.sawCanvas, done: reveal.done, shots: reveal.shots.length },
        stored: rec && { n: rec.n, rule: rec.rule, game: rec.game }, chipsIn: c.inNames.length, chipsOut: c.outCount, survivorsEqual: !!rec && same(c.inNames, rec.survivors) });
      await T.shot(`b3-r${round}-result`);
      await T.tapEl('.pd-mc button', 'รอบต่อไป'); await sleep(900);
    }
    const w = await wd2(T);
    R.renderer = rendererOf(w); R.lost = w.lost; R.console = summarize(T);
    const ok = R.defaultCards && R.halfCards[0].disabled === false && R.halfCards[1].disabled === false && R.halfCards[2].disabled === true && R.halfCards[3].disabled === true
      && R.edited.cards[0].disabled === true && R.edited.cards[0].pressed === 'false' && R.finalPressed[0] === R.finalPressed[1] && R.finalPressed[0] === 'ไพ่แจกทั้งห้อง' && R.rounds.every((x) => x.stored && x.survivorsEqual) && R.rounds[0].stored.n === 20 && R.rounds[1].stored.n === 10;
    R.verdict = ok ? 'PASS' : 'CHECK';
    await T.close();
    save('b3', R);
    return { verdict: R.verdict, halfCards: R.halfCards, edited: R.edited.cards, rounds: R.rounds };
  },
};

async function walk(W, H, mobile) {

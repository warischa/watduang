const b64 = (o) => Buffer.from(JSON.stringify(o)).toString('base64');
function answer(method, path, body) {
  if (method === 'POST' && path === '/api/draw/events') return [201, { eventId: EVENT_ID, pin: PIN, adminToken: ADMIN, expiresAt: new Date(Date.now() + 72 * 3600e3).toISOString() }];
  const m = path.match(/^\/api\/draw\/events\/([^/]+)\/(.*)$/); if (!m) return [404, { error: 'no-route' }];
  const rest = m[2];
  if (method === 'GET' && rest === 'names') return [200, { state: stub.started ? 'started' : 'open', names: PEOPLE.map((p) => ({ ...p, joined: Date.now() })) }];
  if (method === 'POST' && rest === 'start') { stub.started = true; let ids = []; try { ids = JSON.parse(body).ids; } catch {} stub.roster = ids.map((id) => PEOPLE.find((p) => p.id === id)).filter(Boolean).map(({ id, name }) => ({ id, name })); return [200, { roster: stub.roster }]; }
  if (method === 'POST' && rest === 'join') { const j = JSON.parse(body || '{}'); const p = PEOPLE.find((q) => q.id === stub.joinAs) || PEOPLE[0]; return [201, { id: p.id, name: j.name || p.name, eventKey: PIN }]; }
  const s = rest.match(/^status\/(.+)$/);
  if (method === 'GET' && s) {
    if (stub.failStatus) return [503, { error: 'unavailable' }];
    if (stub.statusMode === 'open') return [200, { state: 'open' }];
    return [200, { state: 'started', member: stub.memberIds ? stub.memberIds.includes(s[1]) : true }];
  }
  return [404, { error: 'no-route' }];
}
/** A tab with the stub on. `all` pauses every request (to cut or to log the network). */
export async function stubTab(opts = {}) {
  const T = await openTab(opts);
  T.S.apiLog = [];
  T.S.onEvent = (m) => {
    if (m.method !== 'Fetch.requestPaused') return;
    const p = m.params; const u = new URL(p.request.url); const path = u.pathname;
    const entry = { t: Date.now(), method: p.request.method, path: path.length > 70 ? path.slice(0, 70) : path };
    if (T.cut) { T.S.cutLog.push(entry); T.send('Fetch.failRequest', { requestId: p.requestId, errorReason: 'InternetDisconnected' }); return; }
    if (path.startsWith('/api/draw/')) {
      T.S.apiLog.push(entry);
      const [code, obj] = answer(p.request.method, path, p.request.postData);
      T.send('Fetch.fulfillRequest', { requestId: p.requestId, responseCode: code, responseHeaders: [{ name: 'content-type', value: 'application/json' }, { name: 'cache-control', value: 'no-store' }], body: b64(obj) });
      return;
    }
    T.send('Fetch.continueRequest', { requestId: p.requestId });
  };
  T.S.cutLog = [];
  await T.send('Fetch.enable', { patterns: [{ urlPattern: opts.all ? '*' : '*/api/draw/*' }] });
  return T;
}

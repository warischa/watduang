/** pick the finalist `which` (0 = first listed) onto lane `lane`; the lane tap, then the name tap. */
export async function pickLane(T, lane) {
  await sleep(450);
  let c = await centerN(T, '.pd-lane', lane);
  let blocked = null;
  if (!c) throw new Error(`lane ${lane} absent`);
  if (!c.hit) {
    // record what covers the lane control's centre, then look for any point of the control that does hit it
    blocked = await T.ev(`const e=[...document.querySelectorAll('.pd-lane')][${lane}]; const r=e.getBoundingClientRect(); const top=document.elementFromPoint(${c.x},${c.y}); const d=x=>x?(x.tagName.toLowerCase()+(x.id?'#'+x.id:'')+(x.className&&x.className.baseVal===undefined?'.'+String(x.className).split(' ').join('.'):'')):null; let free=null; for(let fy=0.1;fy<=0.9&&!free;fy+=0.2) for(let fx=0.1;fx<=0.9&&!free;fx+=0.2){const x=r.left+r.width*fx,y=r.top+r.height*fy; const h=document.elementFromPoint(x,y); if(h&&(h===e||e.contains(h))) free={x,y};} return { rect:{l:Math.round(r.left),t:Math.round(r.top),w:Math.round(r.width),h:Math.round(r.height)}, coveredBy:d(top), coveredByRect: top?(()=>{const q=top.getBoundingClientRect(); return {l:Math.round(q.left),t:Math.round(q.top),w:Math.round(q.width),h:Math.round(q.height)}})():null, free };`);
    if (blocked.free) c = { ...c, x: blocked.free.x, y: blocked.free.y };
    else { await T.ev(`[...document.querySelectorAll('.pd-lane')][${lane}].click(); return 1`); }
  }
  if (!blocked || blocked.free) await T.touch(c.x, c.y);
  await sleep(300);
  const nm = await T.ev(`const b=document.querySelector('.pd-name'); return b?b.textContent.trim():null`);
  const n = await T.tapEl('.pd-name', null, { settleMs: 450 });
  await sleep(300);
  return { lane, name: nm, hit: n.hit, blocked };
}

/** one press with screenshots while a 3D canvas is up; names carry the host's own phase. */
export async function pressShots(T, label, prefix, { sel = '.pd-mc button', canvasSel = '.pd-3d canvas', everyMs = 2000, maxShots = 6, maxMs = 150000 } = {}) {
  const w0 = await wd2(T); const t0 = Date.now();
  const tap = await T.tapEl(sel, label, { settleMs: 450 });
  const shots = [], tl = []; let saw = false, last = -1e9, done = false;
  while (Date.now() - t0 < maxMs) {
    const r = await T.ev(`const c=document.querySelector(${JSON.stringify(canvasSel)}); const b=document.querySelector(${JSON.stringify(sel)}); return { canvas:!!c, btn:b?b.textContent.trim():null };`);
    const ms = Date.now() - t0;
    const ph = r.canvas ? await phaseNow(T) : '-';
    if (!tl.length || tl[tl.length - 1].c !== r.canvas || tl[tl.length - 1].b !== r.btn || tl[tl.length - 1].p !== ph) tl.push({ ms, c: r.canvas, b: r.btn, p: ph });
    if (r.canvas) {
      saw = true;
      if (shots.length < maxShots && ms - last >= everyMs) { last = ms; shots.push({ ms, phase: ph, file: await T.shot(`${prefix}-${nn(shots.length + 1)}-${ph}`) }); }
    }
    if (!r.canvas && saw) { await sleep(700); done = true; break; }
    if (!r.canvas && r.btn && r.btn !== label && ms > 4000) { done = true; break; }
    await sleep(80);
  }
  const w1 = await wd2(T);
  return { tap: { disabledAtTap: tap.disabled, hit: tap.hit }, sawCanvas: saw, done, elapsedMs: Date.now() - t0, tl: tl.slice(0, 40), shots, canvasEvents: w1.canvas.length - w0.canvas.length, frames: w1.frames, finalBtn: await mcText(T) };
}

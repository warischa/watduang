# gh#309 UI audit instrument (2026-11-14). Reads painted pixels of the gh309 browser shots; renders nothing.
# Run from anywhere:  python3 -I contrast309.py.txt <repo root>
# Saved as .txt so repo tooling never treats it as code.
#
# What each read counts:
#   text_read  - crop box; ground = the single most common RGB in the box; text = the pixel at the 99.5th
#                percentile of contrast against that ground. Ratio = WCAG 2.x relative luminance.
#   solid_read - mode RGB of box A (a solid fill: outline, icon disc, suit) against mode RGB of box B.
#   extent     - rows of a box holding at least one pixel that passes a colour test: glyph box height in px.
#   band       - Thai base band: rows whose count of text pixels is >= 50% of the busiest row's count.
#   column     - run-length colour classes down one pixel column (gaps, fold, header band).
import os
import sys
from collections import Counter

from PIL import Image

ROOT = sys.argv[1] if len(sys.argv) > 1 else '.'
S9 = 'docs/verification/evidence/gh309-browser-2026-11-14/shots/'

_cache = {}


def load(rel):
    full = os.path.join(ROOT, rel)
    if not os.path.isfile(full):
        raise SystemExit('MISSING ' + rel)
    if rel not in _cache:
        _cache[rel] = Image.open(full).convert('RGB')
    return _cache[rel]


def shot(name):
    return load(S9 + name + '.png')


def lin(c):
    c = c / 255
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def lum(p):
    return 0.2126 * lin(p[0]) + 0.7152 * lin(p[1]) + 0.0722 * lin(p[2])


def ratio(a, b):
    la, lb = lum(a), lum(b)
    return (max(la, lb) + 0.05) / (min(la, lb) + 0.05)


def hx(p):
    return '#%02x%02x%02x' % p[:3]


def h2p(s):
    return tuple(int(s[i:i + 2], 16) for i in (1, 3, 5))


def pixels(img, box):
    x0, y0, x1, y1 = box
    if not (0 <= x0 < x1 <= img.width and 0 <= y0 < y1 <= img.height):
        raise SystemExit('BAD BOX %r on %dx%d' % (box, img.width, img.height))
    return list(img.crop(box).getdata())


def mode(img, box):
    return Counter(pixels(img, box)).most_common(1)[0][0]


def text_read(img, box):
    px = pixels(img, box)
    ground = Counter(px).most_common(1)[0][0]
    ranked = sorted(px, key=lambda p: ratio(p, ground))
    text = ranked[min(len(ranked) - 1, int(0.995 * len(ranked)))]
    return ground, text, ratio(text, ground)


def extent(img, box, test):
    x0, y0, x1, y1 = box
    rows = [y for y in range(y0, y1) if any(test(img.getpixel((x, y))) for x in range(x0, x1))]
    if not rows:
        return None
    return rows[0], rows[-1], rows[-1] - rows[0] + 1


def band(img, box, test):
    x0, y0, x1, y1 = box
    counts = [(y, sum(1 for x in range(x0, x1) if test(img.getpixel((x, y))))) for y in range(y0, y1)]
    peak = max(c for _, c in counts)
    rows = [y for y, c in counts if c >= 0.5 * peak]
    return rows[0], rows[-1], rows[-1] - rows[0] + 1, peak


def near(p, q, d=40):
    return sum(abs(a - b) for a, b in zip(p, q)) <= d


def column(img, x, y0, y1, classes):
    runs = []
    for y in range(y0, y1):
        p = img.getpixel((x, y))
        name = next((n for n, c in classes if near(p, c)), 'other')
        if runs and runs[-1][0] == name:
            runs[-1][2] = y
        else:
            runs.append([name, y, y])
    return [(n, a, b, b - a + 1) for n, a, b in runs]


def gold(p):
    return p[0] > 180 and p[1] > 140 and p[2] < 110 and p[0] - p[2] > 100


def dark(p):
    return lum(p) < 0.2


def light(p):
    return lum(p) > 0.45


def ink_text(p):
    return lum(p) < 0.35
